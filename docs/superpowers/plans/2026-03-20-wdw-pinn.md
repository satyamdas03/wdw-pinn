```python
# src/physics.py
import numpy as np
from scipy.integrate import solve_ivp


def wdw_potential(a: np.ndarray) -> np.ndarray:
    """
    Wheeler-DeWitt potential U(a) = a^2 - a^4
    for closed FRW minisuperspace with Λ=3.

    Positive (classically forbidden) for a < 1.
    Negative (classically allowed) for a > 1.
    Turning point at a = 1.
    """
    return a**2 - a**4


def wkb_amplitude(a: np.ndarray, p: float = 0.0) -> np.ndarray:
    """
    WKB amplitude in classically allowed region (a > 1):
        A(a; p) ∝ a^(-(p+1)/2) · |U(a)|^(-1/4)

    Only valid for a > 1 where U(a) < 0.
    """
    U = wdw_potential(a)
    # |U(a)|^(-1/4)
    envelope = np.abs(U) ** (-0.25)
    # a^(-(p+1)/2) prefactor from operator ordering
    ordering_factor = a ** (-(p + 1) / 2)
    # Normalize so that amplitude = 1 at a = 1.1 (arbitrary)
    amp = envelope * ordering_factor
    return amp / amp[0]


def solve_wdw_reference(
    p: float = 0.0,
    a_min: float = 0.01,
    a_max: float = 3.5,
    n_points: int = 500,
) -> tuple[np.ndarray, np.ndarray]:
    """
    Solve the WDW equation as an ODE using scipy (DOP853).

    WDW ODE: Ψ'' + (p/a)Ψ' - U(a)Ψ = 0
    Rewritten as system: y[0] = Ψ, y[1] = Ψ'

    Frobenius series initial conditions at a_min (p=0 case: Ψ(0)=1, Ψ'(0)=0).
    For general p: y0 = [1.0, 0.0] (regular solution branch).

    Returns: (a_grid, Ψ_real) — real-valued solution for validation.
    """
    def rhs(a, y):
        psi, dpsi = y
        a_safe = max(a, 1e-8)
        U = wdw_potential(np.array([a_safe]))[0]
        d2psi = -(p / a_safe) * dpsi + U * psi
        return [dpsi, d2psi]

    a_span = (a_min, a_max)
    y0 = [1.0, 0.0]  # Regular (Hartle-Hawking) branch

    sol = solve_ivp(
        rhs,
        a_span,
        y0,
        method="DOP853",
        dense_output=True,
        rtol=1e-10,
        atol=1e-12,
    )

    a_grid = np.linspace(a_min, a_max, n_points)
    psi = sol.sol(a_grid)[0]
    return a_grid, psi
```

```python
# src/model.py
import torch
import torch.nn as nn
import numpy as np


class SIRENLayer(nn.Module):
    """
    Single sinusoidal layer from SIREN (Sitzmann et al. 2020).
    Activation: sin(omega_0 * Wx + b)

    Key property: derivatives of SIREN are also SIREN networks,
    making PDE residual computation via autograd exact within the function class.
    """

    def __init__(self, in_features: int, out_features: int,
                 omega_0: float = 30.0, is_first: bool = False):
        super().__init__()
        self.omega_0 = omega_0
        self.is_first = is_first
        self.linear = nn.Linear(in_features, out_features)
        self._init_weights()

    def _init_weights(self):
        with torch.no_grad():
            if self.is_first:
                # Uniform(-1/n_in, 1/n_in)
                n = self.linear.weight.shape[1]
                self.linear.weight.uniform_(-1.0 / n, 1.0 / n)
            else:
                # Uniform(-sqrt(6/n_in)/omega_0, sqrt(6/n_in)/omega_0)
                n = self.linear.weight.shape[1]
                bound = np.sqrt(6.0 / n) / self.omega_0
                self.linear.weight.uniform_(-bound, bound)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return torch.sin(self.omega_0 * self.linear(x))


class WDWNet(nn.Module):
    """
    1D PINN for Wheeler-DeWitt equation.
    Input:  a (scale factor), shape [N, 1]
    Output: [Re(Ψ), Im(Ψ)],  shape [N, 2]

    Fixed operator ordering p (set at construction or passed separately).
    Used for validation against scipy reference solver.
    """

    def __init__(self, hidden_dim: int = 256, n_layers: int = 5,
                 omega_0: float = 30.0):
        super().__init__()
        self.first = SIRENLayer(1, hidden_dim, omega_0=omega_0, is_first=True)
        self.hidden = nn.ModuleList([
            SIRENLayer(hidden_dim, hidden_dim, omega_0=omega_0)
            for _ in range(n_layers - 1)
        ])
        self.output = nn.Linear(hidden_dim, 2)

    def forward(self, a: torch.Tensor) -> torch.Tensor:
        x = self.first(a)
        for layer in self.hidden:
            x = layer(x)
        return self.output(x)


class MetaWDWNet(nn.Module):
    """
    Meta-PINN for Wheeler-DeWitt equation over all operator orderings.
    Input:  (a, p) — scale factor + ordering parameter, shape [N, 2]
    Output: [Re(Ψ), Im(Ψ)],  shape [N, 2]

    A single model learns Ψ(a; p) continuously over p ∈ [-2, 4].
    This enables the p-sweep experiment without retraining.
    """

    def __init__(self, hidden_dim: int = 256, n_layers: int = 6,
                 omega_0: float = 30.0):
        super().__init__()
        self.first = SIRENLayer(2, hidden_dim, omega_0=omega_0, is_first=True)
        self.hidden = nn.ModuleList([
            SIRENLayer(hidden_dim, hidden_dim, omega_0=omega_0)
            for _ in range(n_layers - 1)
        ])
        self.output = nn.Linear(hidden_dim, 2)

    def forward(self, a: torch.Tensor, p: torch.Tensor) -> torch.Tensor:
        x = torch.cat([a, p], dim=-1)
        x = self.first(x)
        for layer in self.hidden:
            x = layer(x)
        return self.output(x)
```

```python
# src/loss.py
"""
Loss functions for the Wheeler-DeWitt PINN.

Three terms:
  L_pde  — PDE residual: ||Ψ'' + (p/a)Ψ' - U(a)Ψ||²
  L_bc   — boundary conditions (Hartle-Hawking or Vilenkin)
  L_norm — normalization pin: |Ψ(a_ref) - [1,0]|² (prevents Ψ=0 collapse)
"""
import torch


def _wdw_potential(a: torch.Tensor) -> torch.Tensor:
    """U(a) = a² - a⁴  (turning point at a=1)."""
    return a**2 - a**4


def pde_residual(
    a: torch.Tensor,
    psi_r: torch.Tensor,
    psi_i: torch.Tensor,
    p: float,
) -> torch.Tensor:
    """
    Compute mean squared WDW residual:
        residual = Ψ'' + (p/a)·Ψ' - U(a)·Ψ

    Uses autograd — a must have requires_grad=True.
    Handles singularity at a=0 by clamping denominator.
    """
    U = _wdw_potential(a)
    a_safe = a.clamp(min=1e-6)

    # First derivatives
    dpsi_r = torch.autograd.grad(
        psi_r.sum(), a, create_graph=True, retain_graph=True
    )[0]
    dpsi_i = torch.autograd.grad(
        psi_i.sum(), a, create_graph=True, retain_graph=True
    )[0]

    # Second derivatives
    d2psi_r = torch.autograd.grad(
        dpsi_r.sum(), a, create_graph=True, retain_graph=True
    )[0]
    d2psi_i = torch.autograd.grad(
        dpsi_i.sum(), a, create_graph=True, retain_graph=True
    )[0]

    res_r = d2psi_r + (p / a_safe) * dpsi_r - U * psi_r
    res_i = d2psi_i + (p / a_safe) * dpsi_i - U * psi_i

    return torch.mean(res_r**2 + res_i**2)


def bc_loss_hartle_hawking(
    psi_at_origin: torch.Tensor,      # Ψ(a_min): shape [1, 2]
    dpsi_at_origin: torch.Tensor,     # Ψ'(a_min): shape [1, 2]
    psi_at_amax: torch.Tensor,        # Ψ(a_max): shape [1, 2]
) -> torch.Tensor:
    """
    Hartle-Hawking (no-boundary) boundary conditions:
      1. Regularity at a~0: Ψ'(a_min) = 0
      2. Decay at large a: Ψ(a_max) → 0

    Note: Ψ(a_min) itself is free (it's pinned by normalization loss, not BC).
    """
    # Regularity: derivative at origin must vanish
    loss_reg = torch.sum(dpsi_at_origin**2)
    # Decay: wave function must decay at large a (classically forbidden region)
    loss_decay = torch.sum(psi_at_amax**2)
    return loss_reg + loss_decay


def bc_loss_vilenkin(
    psi_at_amax: torch.Tensor,        # Ψ(a_max): shape [1, 2]
    dpsi_at_amax: torch.Tensor,       # Ψ'(a_max): shape [1, 2]
    a_max: float,
) -> torch.Tensor:
    """
    Vilenkin (tunneling) boundary condition: outgoing wave only at a_max.
    WKB momentum at a_max: k = sqrt(|U(a_max)|) = sqrt(a_max^4 - a_max^2)

    Condition: Ψ' = +i·k·Ψ  (outgoing → expanding universe)
    In real/imag split:
        Re(Ψ)' = -k·Im(Ψ)
        Im(Ψ)' = +k·Re(Ψ)
    """
    U_amax = a_max**2 - a_max**4  # Should be negative for a_max > 1
    k = (-U_amax) ** 0.5  # WKB momentum, real for a_max > 1

    psi_r, psi_i = psi_at_amax[:, 0:1], psi_at_amax[:, 1:2]
    dpsi_r, dpsi_i = dpsi_at_amax[:, 0:1], dpsi_at_amax[:, 1:2]

    res_r = dpsi_r + k * psi_i   # Re(Ψ') + k·Im(Ψ) = 0
    res_i = dpsi_i - k * psi_r   # Im(Ψ') - k·Re(Ψ) = 0

    return torch.mean(res_r**2 + res_i**2)


def normalization_loss(psi_at_ref: torch.Tensor) -> torch.Tensor:
    """
    Pin Ψ(a_ref) = [1, 0] to prevent trivial Ψ=0 solution.
    a_ref is typically 0.1 (deep in classically forbidden region,
    where HH wave function should be real and positive).
    """
    target = torch.tensor([[1.0, 0.0]], dtype=psi_at_ref.dtype,
                           device=psi_at_ref.device)
    return torch.mean((psi_at_ref - target)**2)


def total_loss(
    L_pde: torch.Tensor,
    L_bc: torch.Tensor,
    L_norm: torch.Tensor,
    lambda_pde: float = 1.0,
    lambda_bc: float = 100.0,
    lambda_norm: float = 50.0,
) -> torch.Tensor:
    """
    Weighted sum of all loss terms.
    Default weights: BC dominates early to prevent Ψ=0 collapse,
    then lambda_pde is increased during training via curriculum.
    """
    return lambda_pde * L_pde + lambda_bc * L_bc + lambda_norm * L_norm
```

```python
# src/train.py
"""
Training loop for Wheeler-DeWitt PINN.

Curriculum: expand domain from tunneling region → full domain.
Optimizer:  Adam (rapid convergence) → L-BFGS (tight residuals).
RAR:        adaptive collocation sampling every K steps.
"""
from __future__ import annotations
import torch
import numpy as np
from typing import Callable
from tqdm import tqdm

from src.loss import pde_residual, bc_loss_hartle_hawking, normalization_loss, total_loss


def sample_collocation(a_min: float, a_max: float, n: int) -> torch.Tensor:
    """Uniform random collocation points on [a_min, a_max]."""
    return torch.FloatTensor(n, 1).uniform_(a_min, a_max).requires_grad_(True)


def rar_resample(
    model_fn: Callable,
    a_colloc: torch.Tensor,
    p_val: float,
    n_add: int,
    a_min: float,
    a_max: float,
) -> torch.Tensor:
    """
    Residual-Adaptive Refinement: add collocation points where PDE residual is largest.
    """
    a_dense = torch.linspace(a_min, a_max, 1000).unsqueeze(1).requires_grad_(True)
    psi = model_fn(a_dense)
    psi_r, psi_i = psi[:, 0:1], psi[:, 1:2]
    res = pde_residual(a_dense, psi_r, psi_i, p=p_val)

    # Re-compute per-point residuals for sampling
    with torch.no_grad():
        a_np = torch.linspace(a_min, a_max, 1000).unsqueeze(1)
        a_np.requires_grad_(True)
        psi2 = model_fn(a_np)
        psi_r2, psi_i2 = psi2[:, 0:1], psi2[:, 1:2]
        U = a_np**2 - a_np**4
        per_point = (psi_r2 * U).abs().squeeze().detach().numpy()
        per_point = per_point / per_point.sum()

    a_candidates = np.linspace(a_min, a_max, 1000)
    chosen = np.random.choice(a_candidates, size=n_add, p=per_point)
    new_points = torch.tensor(chosen, dtype=torch.float32).unsqueeze(1).requires_grad_(True)
    return new_points


def train_wdw(
    model,
    p: float = 0.0,
    bc_type: str = "hartle_hawking",
    a_ref: float = 0.1,
    a_max_final: float = 3.5,
    n_colloc: int = 3000,
    n_epochs_adam: int = 5000,
    n_epochs_lbfgs: int = 500,
    lambda_bc: float = 100.0,
    lambda_norm: float = 50.0,
    verbose: bool = True,
) -> list[float]:
    """
    Train a WDWNet for a fixed operator ordering p.

    Returns: list of total loss values per epoch.
    """
    losses = []
    optimizer_adam = torch.optim.Adam(model.parameters(), lr=1e-3)

    # Curriculum: phases defined as (a_max, lambda_pde, n_epochs)
    curriculum = [
        (1.2,          0.1,  n_epochs_adam // 3),
        (2.0,          1.0,  n_epochs_adam // 3),
        (a_max_final,  1.0,  n_epochs_adam - 2 * (n_epochs_adam // 3)),
    ]

    a_min = 0.01

    # --- Adam phase with curriculum ---
    for (a_max_curr, lambda_pde, n_ep) in curriculum:
        a_colloc = sample_collocation(a_min, a_max_curr, n_colloc)
        bar = tqdm(range(n_ep), desc=f"Adam a_max={a_max_curr:.1f}", disable=not verbose)

        for epoch in bar:
            optimizer_adam.zero_grad()
            a_colloc.requires_grad_(True)

            # Forward pass
            psi = model(a_colloc)
            psi_r, psi_i = psi[:, 0:1], psi[:, 1:2]

            # Boundary condition points
            a_bc_min = torch.tensor([[a_min]], requires_grad=True)
            a_bc_max = torch.tensor([[a_max_final]], requires_grad=True)
            a_bc_ref = torch.tensor([[a_ref]], requires_grad=True)

            psi_min = model(a_bc_min)
            psi_max = model(a_bc_max)
            psi_ref = model(a_bc_ref)

            dpsi_min = torch.autograd.grad(
                psi_min.sum(), a_bc_min, create_graph=True
            )[0]

            # Losses
            L_pde = pde_residual(a_colloc, psi_r, psi_i, p=p)
            L_bc = bc_loss_hartle_hawking(psi_min, dpsi_min, psi_max)
            L_norm = normalization_loss(psi_ref)
            L = total_loss(L_pde, L_bc, L_norm, lambda_pde, lambda_bc, lambda_norm)

            L.backward()
            torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
            optimizer_adam.step()

            losses.append(L.item())
            if epoch % 200 == 0:
                bar.set_postfix(loss=f"{L.item():.4e}")

            # RAR every 1000 steps: adaptively add points where residual is largest
            if epoch % 1000 == 999:
                def _model_fn(a_t):
                    return model(a_t)
                new_pts = rar_resample(_model_fn, a_colloc, p, n_add=500,
                                       a_min=a_min, a_max=a_max_curr)
                a_colloc = torch.cat([a_colloc.detach(), new_pts], dim=0)[:n_colloc]
                a_colloc = a_colloc.detach().requires_grad_(True)

    # --- L-BFGS fine-tuning phase ---
    a_colloc = sample_collocation(a_min, a_max_final, n_colloc * 2)
    optimizer_lbfgs = torch.optim.LBFGS(
        model.parameters(), lr=0.1, max_iter=50,
        history_size=50, line_search_fn="strong_wolfe"
    )

    bar = tqdm(range(n_epochs_lbfgs), desc="L-BFGS fine-tune", disable=not verbose)

    for epoch in bar:
        def closure():
            optimizer_lbfgs.zero_grad()
            a_c = a_colloc.detach().requires_grad_(True)
            psi = model(a_c)
            psi_r, psi_i = psi[:, 0:1], psi[:, 1:2]

            a_bc_min = torch.tensor([[a_min]], requires_grad=True)
            a_bc_max = torch.tensor([[a_max_final]], requires_grad=True)
            a_bc_ref = torch.tensor([[a_ref]], requires_grad=True)

            psi_min = model(a_bc_min)
            psi_max = model(a_bc_max)
            psi_ref = model(a_bc_ref)
            dpsi_min = torch.autograd.grad(psi_min.sum(), a_bc_min, create_graph=True)[0]

            L_pde = pde_residual(a_c, psi_r, psi_i, p=p)
            L_bc = bc_loss_hartle_hawking(psi_min, dpsi_min, psi_max)
            L_norm = normalization_loss(psi_ref)
            L = total_loss(L_pde, L_bc, L_norm, 1.0, lambda_bc, lambda_norm)
            L.backward()
            return L

        L = optimizer_lbfgs.step(closure)
        losses.append(L.item())
        if epoch % 50 == 0:
            bar.set_postfix(loss=f"{L.item():.4e}")

    return losses
```

```python
from src.model import MetaWDWNet
from src.loss import pde_residual, bc_loss_hartle_hawking, normalization_loss, total_loss
from src.train import sample_collocation


def train_meta_pinn(
    p_min: float = -2.0,
    p_max: float = 4.0,
    n_epochs: int = 8000,
    hidden_dim: int = 256,
    n_layers: int = 6,
    n_colloc: int = 4000,
    verbose: bool = True,
) -> MetaWDWNet:
    """
    Train MetaWDWNet: Ψ(a; p) over all operator orderings simultaneously.

    Batches sample (a, p) pairs uniformly from their respective ranges.
    A single model learns all orderings continuously.
    """
    model = MetaWDWNet(hidden_dim=hidden_dim, n_layers=n_layers, omega_0=30.0)
    optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
    scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=n_epochs)

    a_min, a_max_final = 0.01, 3.5
    a_ref = 0.1

    curriculum = [
        (1.2,   0.1,  n_epochs // 3),
        (2.0,   0.5,  n_epochs // 3),
        (3.5,   1.0,  n_epochs - 2 * (n_epochs // 3)),
    ]

    bar = tqdm(total=n_epochs, desc="Meta-PINN training", disable=not verbose)
    loss_history = []

    for (a_max_curr, lambda_pde, n_ep) in curriculum:
        for _ in range(n_ep):
            optimizer.zero_grad()

            # Sample collocation points: random a and random p per batch
            a_colloc = torch.FloatTensor(n_colloc, 1).uniform_(a_min, a_max_curr).requires_grad_(True)
            p_batch = torch.FloatTensor(n_colloc, 1).uniform_(p_min, p_max)

            psi = model(a_colloc, p_batch)
            psi_r, psi_i = psi[:, 0:1], psi[:, 1:2]

            # PDE residual — use mean p_batch for the (p/a) term
            # Actually compute per-sample: expand p into residual
            U = a_colloc**2 - a_colloc**4
            a_safe = a_colloc.clamp(min=1e-6)

            dpsi_r = torch.autograd.grad(psi_r.sum(), a_colloc, create_graph=True, retain_graph=True)[0]
            dpsi_i = torch.autograd.grad(psi_i.sum(), a_colloc, create_graph=True, retain_graph=True)[0]
            d2psi_r = torch.autograd.grad(dpsi_r.sum(), a_colloc, create_graph=True, retain_graph=True)[0]
            d2psi_i = torch.autograd.grad(dpsi_i.sum(), a_colloc, create_graph=True, retain_graph=True)[0]

            res_r = d2psi_r + (p_batch / a_safe) * dpsi_r - U * psi_r
            res_i = d2psi_i + (p_batch / a_safe) * dpsi_i - U * psi_i
            L_pde = torch.mean(res_r**2 + res_i**2)

            # BC and norm (evaluate at fixed p values spanning range)
            p_bc_vals = torch.FloatTensor([[-2.0], [0.0], [1.0], [2.0], [4.0]])
            L_bc_total = torch.tensor(0.0)
            L_norm_total = torch.tensor(0.0)

            for p_bc in p_bc_vals:
                p_bc_expand = p_bc.expand(1, 1)
                a_bc_min = torch.tensor([[a_min]], requires_grad=True)
                a_bc_max = torch.tensor([[a_max_final]], requires_grad=True)
                a_bc_ref = torch.tensor([[a_ref]], requires_grad=True)

                psi_min = model(a_bc_min, p_bc_expand)
                psi_max = model(a_bc_max, p_bc_expand)
                psi_ref_ = model(a_bc_ref, p_bc_expand)
                dpsi_min = torch.autograd.grad(psi_min.sum(), a_bc_min, create_graph=True)[0]

                L_bc_total = L_bc_total + bc_loss_hartle_hawking(psi_min, dpsi_min, psi_max)
                L_norm_total = L_norm_total + normalization_loss(psi_ref_)

            L_bc_total = L_bc_total / len(p_bc_vals)
            L_norm_total = L_norm_total / len(p_bc_vals)

            L = total_loss(L_pde, L_bc_total, L_norm_total, lambda_pde, 100.0, 50.0)
            L.backward()
            torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
            optimizer.step()
            scheduler.step()

            loss_history.append(L.item())
            bar.update(1)
            bar.set_postfix(loss=f"{L.item():.4e}")

    bar.close()
    torch.save(model.state_dict(), "results/meta_pinn_checkpoint.pt")
    print("Meta-PINN checkpoint saved: results/meta_pinn_checkpoint.pt")

    # Plot loss curve
    plt.figure(figsize=(8, 4))
    plt.semilogy(loss_history)
    plt.xlabel("Epoch"); plt.ylabel("Total loss")
    plt.title("Meta-PINN training loss")
    plt.savefig("results/meta_pinn_loss.png", dpi=150)
    plt.close()

    return model
```

```python
def run_ordering_sweep(
    model: MetaWDWNet,
    p_values: list[float] = None,
    a_grid_np: np.ndarray = None,
) -> dict:
    """
    Extract physical observables from meta-PINN across operator ordering values.

    Observables per p:
      - amplitude: |Ψ(a)| — wave function envelope
      - phase: arg(Ψ) — WKB phase (oscillatory region)
      - wkb_freq: |dθ/da| — local oscillation frequency
      - tunnel_ratio: mean|Ψ| for a<0.8 / mean|Ψ| for a>1.5
    """
    if p_values is None:
        p_values = np.linspace(-2, 4, 61).tolist()
    if a_grid_np is None:
        a_grid_np = np.linspace(0.01, 3.5, 500)

    a_tensor = torch.tensor(a_grid_np, dtype=torch.float32).unsqueeze(1)
    results = {}

    with torch.no_grad():
        for p_val in p_values:
            p_tensor = torch.full((len(a_grid_np), 1), p_val)
            psi = model(a_tensor, p_tensor).numpy()
            psi_r, psi_i = psi[:, 0], psi[:, 1]
            amplitude = np.sqrt(psi_r**2 + psi_i**2)
            phase = np.unwrap(np.arctan2(psi_i, psi_r))
            wkb_freq = np.abs(np.gradient(phase, a_grid_np))

            mask_forbidden = a_grid_np < 0.8
            mask_allowed = a_grid_np > 1.5
            tunnel_ratio = (
                amplitude[mask_forbidden].mean() /
                (amplitude[mask_allowed].mean() + 1e-12)
            )

            results[p_val] = {
                "amplitude": amplitude,
                "phase": phase,
                "wkb_freq": wkb_freq,
                "tunnel_ratio": tunnel_ratio,
                "psi_r": psi_r,
                "psi_i": psi_i,
            }

    return results
```
